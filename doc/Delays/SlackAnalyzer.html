def visit_Gate(g)
  inputs = g.get_inputs
  ip_timings = inputs.collect { |ip| @timings[ip] }
  max = ip_timings.max
  inputs.each do |ip|
    slack = (max - @timings[ip]) + @slack[g]
    update_slack(ip, slack)
    update_slack(ip.get_source, slack)
  end
  super
end
